-- sql/optimization/dashboard.sql: Top 20 recorridos del año que salen de una comuna (filtro de comuna del dashboard).

select
    origin_station_name,
    destination_station_name,
    destination_comuna_label,
    sum(trips) as trips
from gold.agg_station_od_monthly
where origin_comuna_label = 'Comuna 1'
group by origin_station_name, destination_station_name, destination_comuna_label
order by trips desc
limit 20;
